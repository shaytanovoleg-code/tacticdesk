async def tactics_menu(update, context):

    query = update.callback_query

    await query.answer()

    if not is_allowed(query.from_user.id):

        return

    if not TACTICS_URL:

        await safe_edit(

            query,

            "⚠️ Тактична дошка не налаштована.\n"

            "Додайте TACTICS_URL у .env",

            keyboard(button("⬅️ НАЗАД", "start_menu")),

        )

        return

    rows = [

        [InlineKeyboardButton(

            gap_label("🎯 ТАКТИЧНА ДОШКА"),

            web_app=WebAppInfo(url=TACTICS_URL),

        )],

        [button(gap_label("💾 ЗБЕРЕЖЕНЕ"), "tactics_saved")],

        [button("⬅️ НАЗАД", "start_menu")],

    ]

    await safe_edit(

        query,

        bold("🎯 ТАКТИЧНА ДОШКА") + "\n\n"

        "• Натисніть «Тактична дошка», щоб відкрити майданчик.\n"

        "• Перетягуйте фішки, натискайте ⏩, щоб додати кадр.\n"

        "• ▶ — відтворити анімацію.\n"

        "• 💾 — зберегти комбінацію під назвою.\n\n"

        "Тренер отримає повідомлення про кожне збереження.",

        InlineKeyboardMarkup(rows),

        parse_mode=ParseMode.HTML,

    )



async def tactics_saved(update, context):

    query = update.callback_query

    await query.answer()

    if not is_allowed(query.from_user.id):

        return

    # Список збережених веде сама WebApp через localStorage.

    # Тут просто відкриваємо ту саму дошку, де гравець натисне «Збережене».

    if not TACTICS_URL:

        return

    rows = [

        [InlineKeyboardButton(

            gap_label("💾 ВІДКРИТИ ЗБЕРЕЖЕНЕ"),

            web_app=WebAppInfo(url=TACTICS_URL),

        )],

        [button("⬅️ НАЗАД", "tactics_menu")],

    ]

    await safe_edit(

        query,

        "💾 ЗБЕРЕЖЕНЕ\n\n"

        "Натисніть кнопку нижче — у дошці перейдіть на вкладку «Збережене» "

        "(кнопка вгорі праворуч).",

        InlineKeyboardMarkup(rows),

    )



async def web_app_data_handler(update, context):

    """Приймає дані від тактичної дошки."""

    user = update.effective_user

    try:

        raw = update.effective_message.web_app_data.data

        data = json.loads(raw)

    except Exception as error:

        print("WebApp parse error:", error)

        return


    action = data.get("action")

    if action == "save_tactic":

        name = data.get("name", "(без назви)")

        frames_count = data.get("frames_count", 0)


        # Підтвердження гравцеві

        try:

            await update.message.reply_text(

                f"✅ Комбінацію збережено!\n\n"

                f"📝 Назва: {name}\n"

                f"🎞 Кадрів: {frames_count}",

                reply_markup=start_keyboard(),

            )

        except Exception as error:

            print("Помилка відповіді гравцю:", error)


        # Повідомлення головному адміну

        player_name = get_player_name(user.id, fallback=user.full_name)

        for admin_id in MAIN_ADMIN_IDS:

            try:

                await context.bot.send_message(

                    chat_id=admin_id,

                    text=(

                        "🎯 НОВА ТАКТИЧНА КОМБІНАЦІЯ\n\n"

                        f"👤 Хто додав: {player_name}\n"

                        f"🆔 ID: {user.id}\n"

                        f"📝 Назва: {name}\n"

                        f"🎞 Кадрів: {frames_count}"

                    ),

                )

            except Exception as error:

                print(f"Помилка повідомлення адміну {admin_id}:", error)
