const INITIAL = [

  // Господарі (сині) — нижня половина, атакують вгору

  { id: "h-gk",  team: "gk",   label: "GK", x: 50, y: 92 },

  { id: "h-2",   team: "home", label: "2",  x: 78, y: 75 },

  { id: "h-3",   team: "home", label: "3",  x: 22, y: 58 },

  { id: "h-4",   team: "home", label: "4",  x: 78, y: 58 },

  // Гості (червоні) — верхня половина

  { id: "a-gk",  team: "gk",   label: "GK", x: 50, y: 8  },

  { id: "a-1",   team: "away", label: "1",  x: 22, y: 25 },

  { id: "a-2",   team: "away", label: "2",  x: 78, y: 25 },

  { id: "a-3",   team: "away", label: "3",  x: 22, y: 42 },

  { id: "a-4",   team: "away", label: "4",  x: 78, y: 42 },

  // М'яч

];


let pieces = [];               // поточна розстановка

let frames = [];               // історія кадрів (кроків)

let currentFrame = 0;          // який кадр зараз показано

let isAnimating = false;


const field = document.getElementById("field");

const stepInd = document.getElementById("step-indicator");

const btnBack = document.getElementById("btn-back");

const btnFwd  = document.getElementById("btn-fwd");

const btnPlay = document.getElementById("btn-play");


// ============================================================

// РЕНДЕР ФІШОК

// ============================================================

function render() {

  field.innerHTML = "";

  pieces.forEach(p => {

    const el = document.createElement("div");

    el.className = "piece " + p.team;

    el.dataset.id = p.id;

    el.style.left = p.x + "%";

    el.style.top  = p.y + "%";

    el.textContent = p.label;

    field.appendChild(el);

    attachDrag(el, p);

  });

  updateControls();

}


function clonePieces(src) {

  return src.map(p => ({ ...p }));

}


// ============================================================

// DRAG & DROP (мишка + тач)

// ============================================================

function attachDrag(el, piece) {

  let dragging = false;

  let startX, startY, startPX, startPY;

  let moved = false;


  function pointerDown(e) {

    if (isAnimating) return;

    e.preventDefault();

    dragging = true; moved = false;

    const pt = getPoint(e);

    startX = pt.x; startY = pt.y;

    startPX = piece.x; startPY = piece.y;

    el.style.zIndex = 999;

  }


  function pointerMove(e) {

    if (!dragging) return;

    e.preventDefault();

    const pt = getPoint(e);

    const rect = field.getBoundingClientRect();

    const dx = ((pt.x - startX) / rect.width) * 100;

    const dy = ((pt.y - startY) / rect.height) * 100;

    if (Math.abs(dx) > 0.3 || Math.abs(dy) > 0.3) moved = true;

    let nx = clamp(startPX + dx, 3, 97);

    let ny = clamp(startPY + dy, 2, 98);

    piece.x = nx; piece.y = ny;

    el.style.left = nx + "%";

    el.style.top  = ny + "%";

  }


  function pointerUp(e) {

    if (!dragging) return;

    dragging = false;

    el.style.zIndex = "";

    // Фіксуємо кадр після перетягування

    if (moved) {

      saveCurrentFrame();

    }

  }


  el.addEventListener("mousedown", pointerDown);

  el.addEventListener("touchstart", pointerDown, { passive: false });

  document.addEventListener("mousemove", pointerMove);

  document.addEventListener("touchmove", pointerMove, { passive: false });

  document.addEventListener("mouseup", pointerUp);

  document.addEventListener("touchend", pointerUp);

}


function getPoint(e) {

  if (e.touches && e.touches[0]) return { x: e.touches[0].clientX, y: e.touches[0].clientY };

  if (e.changedTouches && e.changedTouches[0]) return { x: e.changedTouches[0].clientX, y: e.changedTouches[0].clientY };

  return { x: e.clientX, y: e.clientY };

}


function clamp(v, min, max) { return Math.max(min, Math.min(max, v)); }


// ============================================================

// КАДРИ (КРОКИ)

// ============================================================

function saveCurrentFrame() {

  // Обрізаємо всі кадри після поточного і додаємо новий

  frames = frames.slice(0, currentFrame + 1);

  frames.push(clonePieces(pieces));

  currentFrame = frames.length - 1;

  updateControls();

}


function gotoFrame(idx, animate = false) {

  if (idx < 0 || idx >= frames.length) return;

  currentFrame = idx;

  const target = frames[idx];


  if (!animate) {

    pieces = clonePieces(target);

    render();

    return;

  }


  // Анімація: плавно зміщуємо кожну фішку до цільових координат

  isAnimating = true;

  const els = {};

  field.querySelectorAll(".piece").forEach(el => { els[el.dataset.id] = el; });


  target.forEach(t => {

    const el = els[t.id];

    if (!el) return;

    el.classList.add("animated");

    el.style.left = t.x + "%";

    el.style.top  = t.y + "%";

  });


  setTimeout(() => {

    pieces = clonePieces(target);

    render();

    isAnimating = false;

  }, 650);

}


function updateControls() {

  stepInd.textContent = (frames.length ? currentFrame + 1 : 0) + " / " + frames.length;

  btnBack.disabled = currentFrame <= 0;

  btnFwd.disabled  = currentFrame >= frames.length - 1;

  btnPlay.disabled = frames.length < 2;

}


// ============================================================

// КНОПКИ

// ============================================================

btnBack.onclick = () => gotoFrame(currentFrame - 1, false);

btnFwd.onclick  = () => {

  if (currentFrame < frames.length - 1) {

    gotoFrame(currentFrame + 1, false);

  } else {

    // Додаємо новий кадр = копія поточного (гравець далі рухає фішку)

    frames.push(clonePieces(pieces));

    currentFrame = frames.length - 1;

    updateControls();

  }

};


btnPlay.onclick = async () => {

  if (frames.length < 2 || isAnimating) return;

  // Відтворюємо від першого до останнього

  gotoFrame(0, false);

  await sleep(200);

  for (let i = 1; i < frames.length; i++) {

    gotoFrame(i, true);

    await sleep(750);

  }

};


function sleep(ms) { return new Promise(r => setTimeout(r, ms)); }


// ============================================================

// ЗБЕРЕЖЕННЯ

// ============================================================

const saveModal = document.getElementById("save-modal");

const saveName  = document.getElementById("save-name");


document.getElementById("btn-save").onclick = () => {

  if (frames.length === 0) { alert("Спочатку щось намалюйте"); return; }

  saveName.value = "";

  saveModal.classList.add("show");

  saveName.focus();

};

document.getElementById("save-cancel").onclick = () => saveModal.classList.remove("show");


document.getElementById("save-ok").onclick = () => {

  const name = saveName.value.trim();

  if (!name) { alert("Введіть назву"); return; }

  const data = {

    name,

    frames: frames,

    initial: INITIAL,

    ts: Date.now(),

  };

  saveLocal(data);

  sendToBot(data);

  saveModal.classList.remove("show");

  alert("✅ Збережено: " + name);

};


function saveLocal(item) {

  const key = "tactics_saved";

  const list = JSON.parse(localStorage.getItem(key) || "[]");

  list.push(item);

  localStorage.setItem(key, JSON.stringify(list));

}


function loadLocal() {

  return JSON.parse(localStorage.getItem("tactics_saved") || "[]");

}


// Відправка головному адміну через Telegram WebApp

function sendToBot(item) {

  if (window.Telegram && Telegram.WebApp) {

    try {

      Telegram.WebApp.sendData(JSON.stringify({

        action: "save_tactic",

        name: item.name,

        frames_count: item.frames.length,

        frames: item.frames,

      }));

    } catch (e) {

      console.warn("sendData error:", e);

    }

  }

}

function renderSaved() {

  const list = loadLocal();

  savedList.innerHTML = "";

  if (!list.length) {

    savedList.innerHTML = '<div style="text-align:center;color:#9be49b;padding:30px 0;">Поки немає збережених комбінацій</div>';

    return;

  }

  // Новіші зверху

  list.slice().reverse().forEach((item, idx) => {

    const div = document.createElement("div");

    div.className = "saved-item";

    const date = new Date(item.ts).toLocaleString("uk-UA");

    div.innerHTML = `

      <div>

        <div class="name">${escapeHtml(item.name)}</div>

        <div class="meta">Кадрів: ${item.frames.length} · ${date}</div>

      </div>

      <button>▶ Відкрити</button>

    `;

    div.querySelector("button").onclick = () => loadSaved(item);

    savedList.appendChild(div);

  });

}


function loadSaved(item) {

  frames = item.frames.map(f => f.map(p => ({ ...p })));

  currentFrame = 0;

  pieces = clonePieces(frames[0]);

  savedView.classList.add("hidden");

  boardView.classList.remove("hidden");

  render();

}


function escapeHtml(s) {

  return s.replace(/[&<>"']/g, c => ({ "&":"&amp;","<":"&lt;",">":"&gt;",'"':"&quot;","'":"&#39;" }[c]));

}
